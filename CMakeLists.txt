cmake_minimum_required(VERSION 3.20)
project(leasequeue VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer (separate build)" OFF)
if(ENABLE_SANITIZERS AND ENABLE_TSAN)
  message(FATAL_ERROR "Use separate sanitizer builds")
endif()
find_package(Threads REQUIRED)
add_library(leasequeue src/queue.cpp)
target_include_directories(leasequeue PUBLIC include)
target_link_libraries(leasequeue PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(leasequeue PUBLIC /W4 /WX /permissive-)
else()
  target_compile_options(leasequeue PUBLIC -Wall -Wextra -Wpedantic -Werror)
  if(ENABLE_SANITIZERS)
    target_compile_options(leasequeue PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(leasequeue PUBLIC -fsanitize=address,undefined)
  endif()
  if(ENABLE_TSAN)
    target_compile_options(leasequeue PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(leasequeue PUBLIC -fsanitize=thread)
  endif()
endif()
add_executable(queue_demo src/demo.cpp)
target_link_libraries(queue_demo PRIVATE leasequeue)
enable_testing()
add_executable(queue_tests tests/queue_tests.cpp)
target_link_libraries(queue_tests PRIVATE leasequeue)
add_test(NAME queue_contract COMMAND queue_tests)
set_tests_properties(queue_contract PROPERTIES TIMEOUT 60)
